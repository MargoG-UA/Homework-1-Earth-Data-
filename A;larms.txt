const resourceId = "e1216fe6-7cbd-41ad-b478-85983a2e2669";
const url = "https://data.kyivcity.gov.ua/api/action/datastore_search?resource_id=" + resourceId;

fetch(url)
    .then(response => {
        if (!response.ok) {
            throw new Error("Failed to fetch data");
        }
        return response.json();
    })
    .then(data => {
        console.log(data);
    })
    .catch(error => {
        console.error("Error fetching data:", error);
    });
